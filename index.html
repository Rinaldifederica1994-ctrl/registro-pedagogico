function findAgendaItem(id){ // Cerca prima nei bambini for(const child of data.children){ const item = child.schedule?.find(s => s.id === id); if(item) return { type: "child", owner: child, item }; } // Poi nei commitments generali const item = data.commitments.find(x => x.id === id); if(item) return { type: "commitment", owner: null, item }; return null; } function deleteAgendaItem(id){ const res = findAgendaItem(id); if(!res) return; const { type, owner } = res; if(type === "child"){ owner.schedule = owner.schedule.filter(s => s.id !== id); } else { data.commitments = data.commitments.filter(x => x.id !== id); } } function openEditAgenda(id){ const res = findAgendaItem(id); if(!res) return; const { type, owner, item } = res; const locationField = type === "child" ? ` ` : ""; modal(`

✏️ Modifica appuntamento

${type==="child" ? esc(owner.name) + " · " + locationIcon(item.location || owner.location) : esc(item.name)}
${locationField}
`); // Eventi document.getElementById("cancelModal").onclick = closeModal; document.getElementById("saveEdit").onclick = () => { const s = document.getElementById("eStart").value; const e = document.getElementById("eEnd").value; // Validazione robusta const start = new Date(`1970-01-01T${s}`); const end = new Date(`1970-01-01T${e}`); if(!s || !e || start >= end){ alert("Controlla gli orari."); return; } item.day = Number(document.getElementById("eDay").value); item.start = s; item.end = e; if(type === "child"){ item.location = document.getElementById("eLocation").value; } closeModal(); renderAll(); }; document.getElementById("deleteEdit").onclick = () => { if(confirm("Eliminare questo impegno dall'agenda?")){ deleteAgendaItem(id); closeModal(); renderAll(); } }; }
